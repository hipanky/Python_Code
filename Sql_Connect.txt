2. Insert multiple rows using executemany()

employees = [
    ("Rahul", "IT", 50000),
    ("Amit", "HR", 45000),
    ("Priya", "Finance", 60000),
    ("Neha", "IT", 55000)
]

sql = """
INSERT INTO employees (name, department, salary)
VALUES (%s, %s, %s)
"""

cur.executemany(sql, employees)

conn.commit()
